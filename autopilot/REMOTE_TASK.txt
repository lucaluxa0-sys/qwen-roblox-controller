		if type(value) ~= "table" then
			return value
		end
		assert(depth < 5)
		local result = {}
		for key, child in pairs(value) do
			result[deepCopy(key, depth + 1)] = deepCopy(child, depth + 1)
		end
		return result
	end
	local original = {a = 1, nested = {b = 2}}
	local copied = deepCopy(original, 0)
	copied.nested.b = 9
	assert(original.nested.b == 2)
	assert(copied.nested.b == 9)
end)

runTest("S064", function()
	local function hasCycle(value, visiting, visited)
		if type(value) ~= "table" then
			return false
		end
		if visiting[value] then
			return true
		end
		if visited[value] then
			return false
		end
		visiting[value] = true
		for key, child in pairs(value) do
			if hasCycle(key, visiting, visited) then
				return true
			end
			if hasCycle(child, visiting, visited) then
				return true
			end
		end
		visiting[value] = nil
		visited[value] = true
		return false
	end
	local cyclic = {}
	cyclic.self = cyclic
	assert(hasCycle(cyclic, {}, {}) == true)
	assert(hasCycle({a = {b = 2}}, {}, {}) == false)
end)
