runTest("S036", function()
	local state = "Idle"
	local function transition(eventName)
		if state == "Idle" and eventName == "Start" then
			state = "Running"
		elseif state == "Running" and eventName == "Pause" then
			state = "Paused"
		elseif state == "Paused" and eventName == "Resume" then
			state = "Running"
		elseif state == "Running" and eventName == "Stop" then
			state = "Idle"
		else
			return false
		end
		return true
	end
	assert(transition("Start") and state == "Running")
	assert(transition("Pause") and state == "Paused")
	assert(transition("Resume") and state == "Running")
	assert(transition("Stop") and state == "Idle")
	assert(transition("Resume") == false)
end)
