	Component.__index = Component
	function Component.new()
		return setmetatable({
			started = false,
			destroyed = false,
			event = Instance.new("BindableEvent"),
			connection = nil,
			count = 0,
		}, Component)
	end
	function Component:Start()
		assert(self.started == false)
		self.started = true
		self.connection = self.event.Event:Connect(function()
			self.count += 1
		end)
		self.event:Fire()
	end
	function Component:Destroy()
		if self.destroyed then
			return
		end
		self.destroyed = true
		if self.connection then
			self.connection:Disconnect()
		end
		self.event:Destroy()
	end
	local component = Component.new()
	component:Start()
	task.wait()
	assert(component.started == true)
	assert(component.count == 1)
	component:Destroy()
	assert(component.destroyed == true)
end)
